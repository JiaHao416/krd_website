```bash
#!/usr/bin/env bash
# D-14：從 git 歷史擷取舊官網 token 到變數，只輸出 HTTP 狀態碼；token 不 echo、不寫檔、不進 ps
set -u
cd /Users/howard/KRD/krd_website || exit 1
# 歷史中 PAT 格式去重後應恰為 1 組；只輸出計數
N=$(git log -p --all -S 'api.airtable.com' -- src/services/index.ts \
  | grep -oE '\bpat[A-Za-z0-9]{14}\.[A-Za-z0-9]{64}\b' | sort -u | wc -l | tr -d ' ')
[ "$N" = "1" ] || { echo "擷取異常：去重後 token 數為 $N（預期 1）"; exit 1; }
T=$(git log -p --all -S 'api.airtable.com' -- src/services/index.ts \
  | grep -oE '\bpat[A-Za-z0-9]{14}\.[A-Za-z0-9]{64}\b' | sort -u | head -1)
[ -n "$T" ] || { echo "擷取失敗（變數為空）"; exit 1; }
# printf 為 shell 內建，header 經 stdin 傳給 curl（-H @-），token 不出現在命令列參數
printf 'Authorization: Bearer %s' "$T" \
  | curl -sS -o /dev/null -w '%{http_code}\n' -H @- https://api.airtable.com/v0/meta/whoami
unset T
```
